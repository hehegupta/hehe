def rail_fence_encrypt(text, rails):
    fence = [['\n' for _ in range(len(text))]
             for _ in range(rails)]

    row = 0
    direction = 1

    for col in range(len(text)):
        fence[row][col] = text[col]

        if row == 0:
            direction = 1
        elif row == rails - 1:
            direction = -1

        row += direction

    result = ""

    for i in range(rails):
        for j in range(len(text)):
            if fence[i][j] != '\n':
                result += fence[i][j]

    return result


def rail_fence_decrypt(cipher, rails):
    fence = [['\n' for _ in range(len(cipher))]
             for _ in range(rails)]

    row = 0
    direction = 1

    # Mark zigzag positions
    for col in range(len(cipher)):
        fence[row][col] = '*'

        if row == 0:
            direction = 1
        elif row == rails - 1:
            direction = -1

        row += direction

    # Fill marked positions
    index = 0
    for i in range(rails):
        for j in range(len(cipher)):
            if fence[i][j] == '*' and index < len(cipher):
                fence[i][j] = cipher[index]
                index += 1

    # Read zigzag
    result = ""
    row = 0
    direction = 1

    for col in range(len(cipher)):
        result += fence[row][col]

        if row == 0:
            direction = 1
        elif row == rails - 1:
            direction = -1

        row += direction

    return result


message = input("Enter message: ").replace(" ", "")
rails = int(input("Enter number of rails: "))

encrypted = rail_fence_encrypt(message, rails)
print("Encrypted Text:", encrypted)

decrypted = rail_fence_decrypt(encrypted, rails)
print("Decrypted Text:", decrypted)
